# 1. יצירת סקריפט ההגדרה בתוך הלינוקס החדש
cat << 'EOF' >> ~/.bashrc

# Auto-start Virtual CAN interface
if ! ip link show vcan0 > /dev/null 2>&1; then
    sudo modprobe vcan 2>/dev/null
    sudo ip link add dev vcan0 type vcan 2>/dev/null
    sudo ip link set up vcan0 2>/dev/null
fi
EOF

# 2. טעינה מחדש של ההגדרות כדי שהשינוי ייכנס לתוקף מייד
source ~/.bashrc

echo "הסקריפט הורץ בהצלחה! מנגנון ה-vcan הוגדר אוטומטית בכל הפעלה."
